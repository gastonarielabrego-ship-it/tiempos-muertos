    94→function fmtDur(s: number): string {
    95→  if (s < 0) return '—';
    96→  const h = Math.floor(s / 3600);
    97→  const m = Math.floor((s % 3600) / 60);
    98→  const sec = s % 60;
    99→  if (h > 0) return `${h}h ${m}m ${sec}s`;
   100→  if (m > 0) return `${m}m ${sec}s`;
   101→  return `${sec}s`;
   102→}

   104→function TurnoBadge({ turno }: { turno: string }) {
   105→  const config: Record<string, { bg: string; text: string; icon: React.ReactNode; label: string }> = {
   106→    TM: { bg: 'bg-amber-100', text: 'text-amber-800', icon: <Sun className="h-3 w-3" />, label: 'TM' },
   107→    TT: { bg: 'bg-orange-100', text: 'text-orange-800', icon: <Sunset className="h-3 w-3" />, label: 'TT' },
   108→    TN: { bg: 'bg-indigo-100', text: 'text-indigo-800', icon: <Moon className="h-3 w-3" />, label: 'TN' },
   109→  };
   110→  const c = config[turno] || config.TM;
   111→  return (
   112→    <span className={`inline-flex items-center gap-0.5 px-1.5 py-0.5 rounded text-[10px] font-bold ${c.bg} ${c.text}`}>
   113→      {c.icon} {c.label}
   114→    </span>
   115→  );
   116→}

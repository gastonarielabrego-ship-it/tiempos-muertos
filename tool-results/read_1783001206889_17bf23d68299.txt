    95→    95→function fmtDur(s: number): string {
    96→    96→  if (s < 0) return '—';
    97→    97→  const h = Math.floor(s / 3600);
    98→    98→  const m = Math.floor((s % 3600) / 60);
    99→    99→  const sec = s % 60;
   100→   100→  if (h > 0) return `${h}h ${m}m ${sec}s`;
   101→   101→  if (m > 0) return `${m}m ${sec}s`;
   102→   102→  return `${sec}s`;
   103→   103→}

   105→   105→function TurnoBadge({ turno }: { turno: string }) {
   106→   106→  const config: Record<string, { bg: string; text: string; icon: React.ReactNode; label: string }> = {
   107→   107→    TM: { bg: 'bg-amber-100', text: 'text-amber-800', icon: <Sun className="h-3 w-3" />, label: 'TM' },
   108→   108→    TT: { bg: 'bg-orange-100', text: 'text-orange-800', icon: <Sunset className="h-3 w-3" />, label: 'TT' },
   109→   109→    TN: { bg: 'bg-indigo-100', text: 'text-indigo-800', icon: <Moon className="h-3 w-3" />, label: 'TN' },
   110→   110→  };
   111→   111→  const c = config[turno] || config.TM;
   112→   112→  return (
   113→   113→    <span className={`inline-flex items-center gap-0.5 px-1.5 py-0.5 rounded text-[10px] font-bold ${c.bg} ${c.text}`}>
   114→   114→      {c.icon} {c.label}
   115→   115→    </span>
   116→   116→  );
   117→   117→}

   105→function fmtDur(s: number): string {
   106→  if (s < 0) return '—';
   107→  const h = Math.floor(s / 3600);
   108→  const m = Math.floor((s % 3600) / 60);
   109→  const sec = s % 60;
   110→  if (h > 0) return `${h}h ${m}m ${sec}s`;
   111→  if (m > 0) return `${m}m ${sec}s`;
   112→  return `${sec}s`;
   113→}

   386→  const handleUpload = async (e: React.ChangeEvent<HTMLInputElement>) => {
   387→    const file = e.target.files?.[0];
   388→    if (!file) return;
   389→    setUploading(true);
   390→    try {
   391→      // Parse xlsx in browser to avoid Vercel payload size limits
   392→      const arrayBuffer = await file.arrayBuffer();
   393→      const workbook = XLSX.read(arrayBuffer, { type: 'array' });
   394→      const sheetName = workbook.SheetNames[0];
   395→      const worksheet = workbook.Sheets[sheetName];
   396→      const rawRows = XLSX.utils.sheet_to_json<Record<string, unknown>>(worksheet);
   397→
   398→      if (!rawRows || rawRows.length === 0) {
   399→        throw new Error('El archivo está vacío');
   400→      }
   401→
   402→      // Validate columns
   403→      const required = ['CODUTI', 'NOMUTI', 'FECHA', 'HORA', 'CODACT', 'ZONSTS', 'ALLSTS', 'DPLSTS', 'NIVSTS', 'CODPRO', 'PCBPRO', 'BULTOS'];
   404→      const missing = required.filter(col => !(col in rawRows[0]));
   405→      if (missing.length > 0) throw new Error(`Faltan columnas: ${missing.join(', ')}`);
   406→
   407→      // Parse all rows on client
   408→      const parsedRows = rawRows.map(row => {
   409→        let fechaStr = '';
   410→        const rawFecha = row['FECHA'];
   411→        if (rawFecha instanceof Date) fechaStr = rawFecha.toISOString();
   412→        else if (typeof rawFecha === 'number') {
   413→          const excelEpoch = new Date(1899, 11, 30);
   414→          fechaStr = new Date(excelEpoch.getTime() + rawFecha * 86400000).toISOString();
   415→        } else if (typeof rawFecha === 'string') fechaStr = new Date(rawFecha).toISOString();
   416→
   417→        let hora = '';
   418→        const rawHora = row['HORA'];
   419→        if (typeof rawHora === 'number') {
   420→          const totalSeconds = Math.floor(rawHora * 86400);
   421→          const h = Math.floor(totalSeconds / 3600);
   422→          const m = Math.floor((totalSeconds % 3600) / 60);
   423→          const s = totalSeconds % 60;
   424→          hora = `${String(h).padStart(2, '0')}:${String(m).padStart(2, '0')}:${String(s).padStart(2, '0')}`;
   425→        } else if (typeof rawHora === 'string') hora = rawHora;
   426→        else if (rawHora instanceof Date) hora = rawHora.toTimeString().slice(0, 8);
   427→
   428→        return {
   429→          codUti: String(row['CODUTI'] || ''),
   430→          nomUti: String(row['NOMUTI'] || ''),
   431→          fecha: fechaStr,
   432→          hora,
   433→          codAct: Number(row['CODACT']) || 0,
   434→          zonSts: row['ZONSTS'] ? String(row['ZONSTS']) : null,
   435→          allSts: Number(row['ALLSTS']) || 0,
   436→          dplSts: Number(row['DPLSTS']) || 0,
   437→          nivSts: Number(row['NIVSTS']) || 0,
   438→          codPro: String(row['CODPRO'] || ''),
   439→          pcbPro: Number(row['PCBPRO']) || 0,
   440→          bultos: Number(row['BULTOS']) || 0,
   441→        };
   442→      });
   443→
   444→      // Send in batches to avoid Vercel payload limits
   445→      const BATCH_SIZE = 2000;
   446→      let totalInserted = 0;
   447→      for (let i = 0; i < parsedRows.length; i += BATCH_SIZE) {
   448→        const batch = parsedRows.slice(i, i + BATCH_SIZE);
   449→        const res = await fetch('/api/upload', {
   450→          method: 'POST',
   451→          headers: { 'Content-Type': 'application/json' },
   452→          body: JSON.stringify({
   453→            rows: batch,
   454→            clearFirst: i === 0,
   455→            totalExpected: parsedRows.length,
   456→          }),
   457→        });
   458→        if (!res.ok) {
   459→          const text = await res.text().catch(() => '');
   460→          throw new Error(text || `Error HTTP ${res.status}`);
   461→        }
   462→        const data = await res.json();
   463→        totalInserted += data.inserted || 0;
   464→      }
   465→
   466→      toast({ title: 'Datos actualizados', description: `${totalInserted} registros cargados` });
   467→      setHasData(true);
   468→      setSelectedOp('all');
   469→      setSelectedTurno('all');
   470→      setSelectedDate('all');
   471→      setActiveTab('ranking');
   472→      await fetchFilters();
   473→    } catch (err) {
   474→      toast({ title: 'Error al cargar', description: err instanceof Error ? err.message : 'Error desconocido', variant: 'destructive' });
   475→    } finally {
   476→      setUploading(false);
   477→      if (fileInputRef.current) fileInputRef.current.value = '';
   478→    }
   479→  };

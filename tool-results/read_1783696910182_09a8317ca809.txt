   104→function fmtDur(s: number): string {
   105→  if (s < 0) return '—';
   106→  const h = Math.floor(s / 3600);
   107→  const m = Math.floor((s % 3600) / 60);
   108→  const sec = s % 60;
   109→  if (h > 0) return `${h}h ${m}m ${sec}s`;
   110→  if (m > 0) return `${m}m ${sec}s`;
   111→  return `${sec}s`;
   112→}

   114→function TurnoBadge({ turno }: { turno: string }) {
   115→  const config: Record<string, { bg: string; text: string; icon: React.ReactNode; label: string }> = {
   116→    TM: { bg: 'bg-amber-100', text: 'text-amber-800', icon: <Sun className="h-3 w-3" />, label: 'TM' },
   117→    TT: { bg: 'bg-orange-100', text: 'text-orange-800', icon: <Sunset className="h-3 w-3" />, label: 'TT' },
   118→    TN: { bg: 'bg-indigo-100', text: 'text-indigo-800', icon: <Moon className="h-3 w-3" />, label: 'TN' },
   119→  };
   120→  const c = config[turno] || config.TM;
   121→  return (
   122→    <span className={`inline-flex items-center gap-0.5 px-1.5 py-0.5 rounded text-[10px] font-bold ${c.bg} ${c.text}`}>
   123→      {c.icon} {c.label}
   124→    </span>
   125→  );
   126→}

   373→      const sheetName = workbook.SheetNames[0];
   374→      const worksheet = workbook.Sheets[sheetName];
   375→      const rawRows = XLSX.utils.sheet_to_json<Record<string, unknown>>(worksheet);
   376→
   377→      if (!rawRows || rawRows.length === 0) {
   378→        throw new Error('El archivo está vacío');
   379→      }
   380→
   381→      // Validate columns
   382→      const required = ['CODUTI', 'NOMUTI', 'FECHA', 'HORA', 'CODACT', 'ZONSTS', 'ALLSTS', 'DPLSTS', 'NIVSTS', 'CODPRO', 'PCBPRO', 'BULTOS'];

   384→      if (missing.length > 0) throw new Error(`Faltan columnas: ${missing.join(', ')}`);

   391→        else if (typeof rawFecha === 'number') {
   392→          const excelEpoch = new Date(1899, 11, 30);
   393→          fechaStr = new Date(excelEpoch.getTime() + rawFecha * 86400000).toISOString();
   394→        } else if (typeof rawFecha === 'string') fechaStr = new Date(rawFecha).toISOString();
   395→
   396→        let hora = '';
   397→        const rawHora = row['HORA'];
   398→        if (typeof rawHora === 'number') {
   399→          const totalSeconds = Math.floor(rawHora * 86400);
   400→          const h = Math.floor(totalSeconds / 3600);
   401→          const m = Math.floor((totalSeconds % 3600) / 60);
   402→          const s = totalSeconds % 60;
   403→          hora = `${String(h).padStart(2, '0')}:${String(m).padStart(2, '0')}:${String(s).padStart(2, '0')}`;
   404→        } else if (typeof rawHora === 'string') hora = rawHora;
   405→        else if (rawHora instanceof Date) hora = rawHora.toTimeString().slice(0, 8);
   406→
   407→        return {
   408→          codUti: String(row['CODUTI'] || ''),
   409→          nomUti: String(row['NOMUTI'] || ''),
   410→          fecha: fechaStr,
   411→          hora,
   412→          codAct: Number(row['CODACT']) || 0,
   413→          zonSts: row['ZONSTS'] ? String(row['ZONSTS']) : null,
   414→          allSts: Number(row['ALLSTS']) || 0,
   415→          dplSts: Number(row['DPLSTS']) || 0,
   416→          nivSts: Number(row['NIVSTS']) || 0,
   417→          codPro: String(row['CODPRO'] || ''),
   418→          pcbPro: Number(row['PCBPRO']) || 0,
   419→          bultos: Number(row['BULTOS']) || 0,
   420→        };
   421→      });
   422→
   423→      // Send in batches to avoid Vercel payload limits
   424→      const BATCH_SIZE = 2000;
   425→      let totalInserted = 0;
   426→      for (let i = 0; i < parsedRows.length; i += BATCH_SIZE) {
   427→        const batch = parsedRows.slice(i, i + BATCH_SIZE);

   429→          method: 'POST',
   430→          headers: { 'Content-Type': 'application/json' },
   431→          body: JSON.stringify({
   432→            rows: batch,
   433→            clearFirst: i === 0,
   434→            totalExpected: parsedRows.length,
   435→          }),
   436→        });

   438→          const text = await res.text().catch(() => '');
   439→          throw new Error(text || `Error HTTP ${res.status}`);
   440→        }
   441→        const data = await res.json();
   442→        totalInserted += data.inserted || 0;
   443→      }

   447→      setSelectedOp('all');
   448→      setSelectedTurno('all');

   453→      toast({ title: 'Error al cargar', description: err instanceof Error ? err.message : 'Error desconocido', variant: 'destructive' });

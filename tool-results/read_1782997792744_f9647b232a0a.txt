    93→function fmtDur(s: number): string {
    94→  if (s < 0) return '—';
    95→  const h = Math.floor(s / 3600);
    96→  const m = Math.floor((s % 3600) / 60);
    97→  const sec = s % 60;
    98→  if (h > 0) return `${h}h ${m}m ${sec}s`;
    99→  if (m > 0) return `${m}m ${sec}s`;
   100→  return `${sec}s`;
   101→}

   103→function TurnoBadge({ turno }: { turno: string }) {
   104→  const config: Record<string, { bg: string; text: string; icon: React.ReactNode; label: string }> = {
   105→    TM: { bg: 'bg-amber-100', text: 'text-amber-800', icon: <Sun className="h-3 w-3" />, label: 'TM' },
   106→    TT: { bg: 'bg-orange-100', text: 'text-orange-800', icon: <Sunset className="h-3 w-3" />, label: 'TT' },
   107→    TN: { bg: 'bg-indigo-100', text: 'text-indigo-800', icon: <Moon className="h-3 w-3" />, label: 'TN' },
   108→  };
   109→  const c = config[turno] || config.TM;
   110→  return (
   111→    <span className={`inline-flex items-center gap-0.5 px-1.5 py-0.5 rounded text-[10px] font-bold ${c.bg} ${c.text}`}>
   112→      {c.icon} {c.label}
   113→    </span>
   114→  );
   115→}

   330→            {/* Filters */}
   331→            <div className="flex flex-wrap items-center gap-2">
   332→              <span className="text-xs font-medium text-muted-foreground">Operador:</span>
   333→              <Select value={selectedOp} onValueChange={handleOpChange}>
   334→                <SelectTrigger className="w-full sm:w-[220px] h-8 text-xs">
   335→                  <SelectValue placeholder="Todos" />
   336→                </SelectTrigger>

   339→                  {filters?.operators.filter(o => o.codUti.trim()).map(o => (

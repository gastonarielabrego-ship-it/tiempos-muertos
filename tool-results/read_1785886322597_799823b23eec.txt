   104→function fmtDur(s: number): string {
   105→  if (s < 0) return '—';
   106→  const h = Math.floor(s / 3600);
   107→  const m = Math.floor((s % 3600) / 60);
   108→  const sec = s % 60;
   109→  if (h > 0) return `${h}h ${m}m ${sec}s`;
   110→  if (m > 0) return `${m}m ${sec}s`;
   111→  return `${sec}s`;
   112→}

   116→    TM: { bg: 'bg-amber-100', text: 'text-amber-800', icon: <Sun className="h-3 w-3" />, label: 'TM' },
   117→    TT: { bg: 'bg-orange-100', text: 'text-orange-800', icon: <Sunset className="h-3 w-3" />, label: 'TT' },
   118→    TN: { bg: 'bg-indigo-100', text: 'text-indigo-800', icon: <Moon className="h-3 w-3" />, label: 'TN' },
   119→  };
   120→  const c = config[turno] || config.TM;
   121→  return (
   122→    <span className={`inline-flex items-center gap-0.5 px-1.5 py-0.5 rounded text-[10px] font-bold ${c.bg} ${c.text}`}>
   123→      {c.icon} {c.label}

   125→  );

   428→  const handleUpload = async (e: React.ChangeEvent<HTMLInputElement>) => {

   430→    if (!file) return;
   431→    setUploading(true);
   432→    try {
   433→      // Parse xlsx in browser to avoid Vercel payload size limits
   434→      const arrayBuffer = await file.arrayBuffer();
   435→      const workbook = XLSX.read(arrayBuffer, { type: 'array' });
   436→      const sheetName = workbook.SheetNames[0];
   437→      const worksheet = workbook.Sheets[sheetName];
   438→      const rawRows = XLSX.utils.sheet_to_json<Record<string, unknown>>(worksheet);
   439→
   440→      if (!rawRows || rawRows.length === 0) {
   441→        throw new Error('El archivo está vacío');
   442→      }
   443→
   444→      // Validate columns
   445→      const required = ['CODUTI', 'NOMUTI', 'FECHA', 'HORA', 'CODACT', 'ZONSTS', 'ALLSTS', 'DPLSTS', 'NIVSTS', 'CODPRO', 'PCBPRO', 'BULTOS'];
   446→      const missing = required.filter(col => !(col in rawRows[0]));
   447→      if (missing.length > 0) throw new Error(`Faltan columnas: ${missing.join(', ')}`);
   448→
   449→      // Parse all rows on client
   450→      const parsedRows = rawRows.map(row => {
   451→        let fechaStr = '';
   452→        const rawFecha = row['FECHA'];
   453→        if (rawFecha instanceof Date) fechaStr = rawFecha.toISOString();
   454→        else if (typeof rawFecha === 'number') {
   455→          const excelEpoch = new Date(1899, 11, 30);
   456→          fechaStr = new Date(excelEpoch.getTime() + rawFecha * 86400000).toISOString();
   457→        } else if (typeof rawFecha === 'string') fechaStr = new Date(rawFecha).toISOString();
   458→
   459→        let hora = '';
   460→        const rawHora = row['HORA'];
   461→        if (typeof rawHora === 'number') {
   462→          const totalSeconds = Math.floor(rawHora * 86400);
   463→          const h = Math.floor(totalSeconds / 3600);
   464→          const m = Math.floor((totalSeconds % 3600) / 60);
   465→          const s = totalSeconds % 60;
   466→          hora = `${String(h).padStart(2, '0')}:${String(m).padStart(2, '0')}:${String(s).padStart(2, '0')}`;
   467→        } else if (typeof rawHora === 'string') hora = rawHora;
   468→        else if (rawHora instanceof Date) hora = rawHora.toTimeString().slice(0, 8);
   469→
   470→        return {
   471→          codUti: String(row['CODUTI'] || ''),
   472→          nomUti: String(row['NOMUTI'] || ''),
   473→          fecha: fechaStr,
   474→          hora,
   475→          codAct: Number(row['CODACT']) || 0,
   476→          zonSts: row['ZONSTS'] ? String(row['ZONSTS']) : null,
   477→          allSts: Number(row['ALLSTS']) || 0,
   478→          dplSts: Number(row['DPLSTS']) || 0,
   479→          nivSts: Number(row['NIVSTS']) || 0,
   480→          codPro: String(row['CODPRO'] || ''),
   481→          pcbPro: Number(row['PCBPRO']) || 0,
   482→          bultos: Number(row['BULTOS']) || 0,
   483→        };
   484→      });
   485→
   486→      // Send in batches to avoid Vercel payload limits
   487→      const BATCH_SIZE = 2000;

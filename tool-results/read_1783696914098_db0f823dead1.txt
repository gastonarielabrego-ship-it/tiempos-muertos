   104→   104→function fmtDur(s: number): string {
   105→   105→  if (s < 0) return '—';
   106→   106→  const h = Math.floor(s / 3600);
   107→   107→  const m = Math.floor((s % 3600) / 60);
   108→   108→  const sec = s % 60;
   109→   109→  if (h > 0) return `${h}h ${m}m ${sec}s`;
   110→   110→  if (m > 0) return `${m}m ${sec}s`;
   111→   111→  return `${sec}s`;
   112→   112→}

   371→   371→      const arrayBuffer = await file.arrayBuffer();

   373→   373→      const sheetName = workbook.SheetNames[0];
   374→   374→      const worksheet = workbook.Sheets[sheetName];
   375→   375→      const rawRows = XLSX.utils.sheet_to_json<Record<string, unknown>>(worksheet);
   376→   376→
   377→   377→      if (!rawRows || rawRows.length === 0) {
   378→   378→        throw new Error('El archivo está vacío');
   379→   379→      }
   380→   380→
   381→   381→      // Validate columns
   382→   382→      const required = ['CODUTI', 'NOMUTI', 'FECHA', 'HORA', 'CODACT', 'ZONSTS', 'ALLSTS', 'DPLSTS', 'NIVSTS', 'CODPRO', 'PCBPRO', 'BULTOS'];
   383→   383→      const missing = required.filter(col => !(col in rawRows[0]));
   384→   384→      if (missing.length > 0) throw new Error(`Faltan columnas: ${missing.join(', ')}`);
   385→   385→
   386→   386→      // Parse all rows on client
   387→   387→      const parsedRows = rawRows.map(row => {
   388→   388→        let fechaStr = '';
   389→   389→        const rawFecha = row['FECHA'];
   390→   390→        if (rawFecha instanceof Date) fechaStr = rawFecha.toISOString();
   391→   391→        else if (typeof rawFecha === 'number') {
   392→   392→          const excelEpoch = new Date(1899, 11, 30);
   393→   393→          fechaStr = new Date(excelEpoch.getTime() + rawFecha * 86400000).toISOString();
   394→   394→        } else if (typeof rawFecha === 'string') fechaStr = new Date(rawFecha).toISOString();
   395→   395→
   396→   396→        let hora = '';
   397→   397→        const rawHora = row['HORA'];
   398→   398→        if (typeof rawHora === 'number') {
   399→   399→          const totalSeconds = Math.floor(rawHora * 86400);
   400→   400→          const h = Math.floor(totalSeconds / 3600);
   401→   401→          const m = Math.floor((totalSeconds % 3600) / 60);
   402→   402→          const s = totalSeconds % 60;
   403→   403→          hora = `${String(h).padStart(2, '0')}:${String(m).padStart(2, '0')}:${String(s).padStart(2, '0')}`;
   404→   404→        } else if (typeof rawHora === 'string') hora = rawHora;
   405→   405→        else if (rawHora instanceof Date) hora = rawHora.toTimeString().slice(0, 8);
   406→   406→
   407→   407→        return {
   408→   408→          codUti: String(row['CODUTI'] || ''),
   409→   409→          nomUti: String(row['NOMUTI'] || ''),
   410→   410→          fecha: fechaStr,
   411→   411→          hora,
   412→   412→          codAct: Number(row['CODACT']) || 0,
   413→   413→          zonSts: row['ZONSTS'] ? String(row['ZONSTS']) : null,
   414→   414→          allSts: Number(row['ALLSTS']) || 0,
   415→   415→          dplSts: Number(row['DPLSTS']) || 0,
   416→   416→          nivSts: Number(row['NIVSTS']) || 0,
   417→   417→          codPro: String(row['CODPRO'] || ''),
   418→   418→          pcbPro: Number(row['PCBPRO']) || 0,
   419→   419→          bultos: Number(row['BULTOS']) || 0,
   420→   420→        };
   421→   421→      });
   422→   422→
   423→   423→      // Send in batches to avoid Vercel payload limits
   424→   424→      const BATCH_SIZE = 2000;
   425→   425→      let totalInserted = 0;
   426→   426→      for (let i = 0; i < parsedRows.length; i += BATCH_SIZE) {
   427→   427→        const batch = parsedRows.slice(i, i + BATCH_SIZE);
   428→   428→        const res = await fetch('/api/upload', {

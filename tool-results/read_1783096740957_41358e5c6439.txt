   100→function fmtDur(s: number): string {
   101→  if (s < 0) return '—';
   102→  const h = Math.floor(s / 3600);
   103→  const m = Math.floor((s % 3600) / 60);
   104→  const sec = s % 60;
   105→  if (h > 0) return `${h}h ${m}m ${sec}s`;
   106→  if (m > 0) return `${m}m ${sec}s`;
   107→  return `${sec}s`;
   108→}

   110→function TurnoBadge({ turno }: { turno: string }) {
   111→  const config: Record<string, { bg: string; text: string; icon: React.ReactNode; label: string }> = {
   112→    TM: { bg: 'bg-amber-100', text: 'text-amber-800', icon: <Sun className="h-3 w-3" />, label: 'TM' },
   113→    TT: { bg: 'bg-orange-100', text: 'text-orange-800', icon: <Sunset className="h-3 w-3" />, label: 'TT' },
   114→    TN: { bg: 'bg-indigo-100', text: 'text-indigo-800', icon: <Moon className="h-3 w-3" />, label: 'TN' },
   115→  };
   116→  const c = config[turno] || config.TM;
   117→  return (
   118→    <span className={`inline-flex items-center gap-0.5 px-1.5 py-0.5 rounded text-[10px] font-bold ${c.bg} ${c.text}`}>
   119→      {c.icon} {c.label}
   120→    </span>
   121→  );
   122→}
